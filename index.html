Cloud Firestore – persistens

const USER_COLLECTION = "users";

export function saveUserData(userId, data) {
  if (!userId) {
    return Promise.resolve();
  }

  const userRef = doc(db, USER_COLLECTION, userId);
  return setDoc(userRef, data, { merge: true });
}

export function loadUserData(userId) {
  if (!userId) {
    return Promise.resolve(null);
  }

  const userRef = doc(db, USER_COLLECTION, userId);

  return getDoc(userRef).then(function (snapshot) {
    if (!snapshot.exists()) {
      return null;
    }

    return snapshot.data();
  });
}

Jag implementerade persistenslagret för att spara och läsa användarspecifik data i Firestore. Funktionen saveUserData sparar data i den inloggade användarens dokument, medan loadUserData hämtar tidigare sparad data när användaren återvänder till applikationen.

Kodexempel 2: Användarspecifik rutthistorik


export async function saveRouteHistory(route) {
  const user = auth.currentUser;
  if (!user) throw new Error("No logged-in user");

  const historyRef = collection(
    db,
    "users",
    user.uid,
    "routeHistory"
  );

  await addDoc(historyRef, {
    routeHash: route.routeHash,
    distance: route.distance,
    createdAt: serverTimestamp()
  });
}

export async function getRouteHistory() {
  const user = auth.currentUser;
  if (!user) throw new Error("No logged-in user");

  const historyRef = collection(
    db,
    "users",
    user.uid,
    "routeHistory"
  );

  const snapshot = await getDocs(historyRef);

  return snapshot.docs.map((doc) => ({
    id: doc.id,
    ...doc.data()
  }));
}

Jag implementerade funktionalitet för att spara och läsa användarens rutthistorik i Cloud Firestore. Rutterna lagras under den inloggade användarens unika ID tillsammans med bland annat ruttens hash, distans och tidpunkt. På så sätt kopplas varje användares rutthistorik till rätt konto.
